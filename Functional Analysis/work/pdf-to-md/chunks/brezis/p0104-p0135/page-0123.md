$$
\rho _ { n } \in C _ { c } ^ { \infty } ( \mathbb { R } ^ { N } ) , \quad \operatorname { \mathrm { s u p p } } \rho _ { n } \subset \overline { { B ( 0 , 1 / n ) } } , \quad \int \rho _ { n } = 1 ,   \rho _ { n } \geq 0   \mathrm { o n }   \mathbb { R } ^ { N } .
